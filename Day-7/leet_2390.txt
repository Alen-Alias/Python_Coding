class Solution:
    def removeStars(self, s: str) -> str:
        def removeHash(s1):
            st=[]
            for i in range(len(s1)):
                if s1[i]!="*":
                    st.append(s1[i])
                else:
                    if st:
                        st.pop()
            return st
        s=removeHash(s)
        return "".join(s)
