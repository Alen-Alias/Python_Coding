# Definition for singly-linked list.
# class ListNode:
#     def __init__(self, val=0, next=None):
#         self.val = val
#         self.next = next
class Solution:
    def middleNode(self, head: Optional[ListNode]) -> Optional[ListNode]:
        temp=head
        l=0
        while temp:
            l+=1
            temp=temp.next
        l=l//2
        temp=head
        while temp.next and l!=0:
            l-=1
            temp=temp.next
        return temp

class Solution:
    def middleNode(self, head: Optional[ListNode]) -> Optional[ListNode]:
        slow,fast=head,head

            fast=fast.next.next
        return slow
