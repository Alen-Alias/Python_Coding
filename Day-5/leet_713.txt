class Solution:
    def numSubarrayProductLessThanK(self, nums: List[int], k: int) -> int:
        m,p,l,r=0,1,0,0
        if k<=1:
            return 0
        while r<len(nums):
            p*=nums[r]
            while p>=k:
                p//=nums[l]
                l+=1
            m+=(r-l+1)
            r+=1
        return m
