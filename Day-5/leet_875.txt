class Solution:
    def minEatingSpeed(self, piles: List[int], h: int) -> int:
        def can_eat(n,piles,h):
            total=0
            for i in piles:
                total+=math.ceil(i/n)
            if total>h:
                return False
            else:
                return True
        s,e=1,max(piles)
        while s<=e:
            mid=(s+e)//2
            if can_eat(mid,piles,h):
                e=mid-1
            else:
                s=mid+1
        return s
