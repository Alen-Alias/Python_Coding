class Solution:
    def lengthOfLongestSubstring(self, s: str) -> int:
        res=""
        l,r=0,0
        m=0
        while r<len(s):
            if s[r] not in res:
                res+=s[r]
                r+=1
                if m<len(res):
                    m=len(res)
            else:
                res=res[1:len(res)]
                l+=1
        return m
