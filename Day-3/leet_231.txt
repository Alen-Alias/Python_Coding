class Solution:
    def isPowerOfTwo(self, n: int) -> bool:
        i=0
        while 2**i <= n:
            if 2**i == n:
                return True
            i+=1
        return False

class Solution:
    def isPowerOfTwo(self, n: int) -> bool:
        if n<1:
            return False
        if n&(n-1)==0:
            return True
        return False
