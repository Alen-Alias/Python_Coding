class Solution:
    def missingNumber(self, nums: List[int]) -> int:
        n=int((len(nums)*(len(nums)+1))/2)
        for i in nums:
            n-=i
        return n

class Solution:
    def missingNumber(self, nums: List[int]) -> int:
        n=int((len(nums)*(len(nums)+1))/2)
        return n-sum(nums)
