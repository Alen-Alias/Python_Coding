class Solution:
    def findMaxAverage(self, l: List[int], k: int) -> float:
        s=sum(l[:k])
        avg=s/k
        max=avg
        for i in range(1,len(l)-k+1):
            s=(s-l[i-1]+l[i+k-1])
            avg=s/k
            if avg>max:
                max=avg
        return max
