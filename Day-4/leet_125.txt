class Solution:
    def isPalindrome(self, s: str) -> bool:
        p=s.lower()
        a=list(p)
        b=[]
        for i in a:
            if (i.isalnum()):
                b.append(i)
        i=0
        j=len(b)-1

        while i<j:
            if b[i]!=b[j]:
                return False
            i+=1
            j-=1
        return True
