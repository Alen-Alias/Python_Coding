class Solution:
    def leftRightDifference(self, nums: List[int]) -> List[int]:
        left=[]
        right=[]
        sum=0
        for i in nums:
            left.append(sum)
            sum+=i
        sum=0
        for i in reversed(nums):
            right.append(sum)
            sum+=i
        right=right[::-1]
        ans=[]
        for i in range(len(nums)):
            ans.append(abs(left[i]-right[i]))
        return ans
