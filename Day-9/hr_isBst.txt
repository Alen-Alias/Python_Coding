class node:
    def __init__(self, data):
        self.data = data
        self.left = None
        self.right = None

def newNode():
    temp = node(-1)
    temp.left = None
    temp.right = None
    return(temp);

def check_binary_search_tree_(root):
    a=[]
    def inOrder(root):
        if root is None:
            return 0
        inOrder(root.left)
        a.append(root.data)
        inOrder(root.right)
        return a
    inOrder(root)
    for i in range(len(a)-1):
        if a[i]>=a[i+1]:
            return False
    return True
